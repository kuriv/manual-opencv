# 图像直方图绘制

图像直方图不再表征任何的图像纹理信息，而是对图像像素的统计。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img = imread("demo.jpg");
    Mat gray;
    Mat hist;
    if (img.empty())
    {
        cout << "could not load image." << endl;
        return -1;
    }
    imshow("IMG", img);
    cvtColor(img, gray, COLOR_BGR2GRAY);
    imshow("GRAY", gray);
    const int channels[] = {0};
    const int bins[] = {256};
    float inRanges[] = {0, 255};
    const float *ranges[] = {inRanges};
    calcHist(&gray, 1, channels, Mat(), hist, 1, bins, ranges);
    cout << hist << endl;
    int hist_w = 512;
    int hist_h = 600;
    int width = 2;
    Mat hist_img = Mat::zeros(hist_h, hist_w, CV_8UC3);
    for (int i = 1; i < hist.rows; i++)
    {
        rectangle(hist_img, Point(width * (i - 1), hist_h - 1),
                  Point(width * i - 1, hist_h - cvRound(hist.at<float>(i - 1) / 15)),
                  Scalar(255, 255, 255), -1);
    }
    imshow("HIST_IMG", hist_img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```
