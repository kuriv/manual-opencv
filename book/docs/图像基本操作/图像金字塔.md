# 图像金字塔

图像金字塔是同一张图片不同分辨率的子图的合集。执行下面的代码，构建高斯金字塔。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img;
    vector<Mat> gauss;
    int level = 3;
    Mat layer;
    img = imread("demo.jpg");
    if (img.empty())
    {
        cout << "Could not load image." << endl;
        return -1;
    }
    gauss.push_back(img);
    for (int i = 0; i < level; i++)
    {
        pyrDown(gauss[i], layer);
        gauss.push_back(layer);
    }
    for (int i = 0; i < gauss.size(); i++)
    {
        imshow("GAUSS" + to_string(i), gauss[i]);
    }
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```
