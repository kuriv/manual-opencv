# 绘制几何图形

执行下面的代码，在图像矩阵上绘制矩形。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img(512, 512, CV_8UC3, Scalar(0, 255, 255));
    Rect rect;
    rect.x = 100;
    rect.y = 100;
    rect.width = 200;
    rect.height = 100;
    rectangle(img, rect, Scalar(255, 255, 0), -1);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，在图像矩阵上绘制圆形。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img(512, 512, CV_8UC3, Scalar(0, 255, 255));
    circle(img, Point(256, 256), 100, Scalar(255, 255, 0), -1);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，在图像矩阵上绘制线段。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])

```

执行下面的代码，在图像矩阵上绘制椭圆。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img(512, 512, CV_8UC3, Scalar(0, 255, 255));
    RotatedRect rr;
    rr.center = Point(256, 256);
    rr.size = Size(200, 100);
    rr.angle = 0;
    ellipse(img, rr, Scalar(255, 255, 0), -1);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```
