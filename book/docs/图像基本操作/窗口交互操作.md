# 窗口交互操作

执行下面的代码，通过操作滚动条实时调整图像的亮度。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

Mat img, result;
int brightness = 50;

void callback_brightness(int, void *)
{
    brightness -= 50;
    cout << "Brightness: " << brightness << endl;
    result = img + Scalar(brightness, brightness, brightness);
    imshow("IMG", result);
}

int main(int argc, char const *argv[])
{
    img = imread("demo.jpg");
    if (img.empty())
    {
        cout << "could not load image." << endl;
        return -1;
    }
    imshow("IMG", img);
    createTrackbar("Brightness: ", "IMG", &brightness, 100, callback_brightness);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，实现同时调整亮度与对比度。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

Mat img, result;
int brightness = 50;
int contrast = 50;

void callback_brightness(int, void *)
{
    brightness -= 50;
    cout << "Brightness: " << brightness << ", Contrast: " << (contrast / 50.0) << endl;
    result = (img + Scalar(brightness, brightness, brightness)) * (contrast / 50.0);
    imshow("IMG", result);
}

void callback_contrast(int, void *)
{
    cout << "Brightness: " << brightness << ", Contrast: " << (contrast / 50.0) << endl;
    result = (img + Scalar(brightness, brightness, brightness)) * (contrast / 50.0);
    imshow("IMG", result);
}

int main(int argc, char const *argv[])
{
    img = imread("demo.jpg");
    if (img.empty())
    {
        cout << "could not load image." << endl;
        return -1;
    }
    imshow("IMG", img);
    createTrackbar("Brightness: ", "IMG", &brightness, 100, callback_brightness);
    createTrackbar("Contrast: ", "IMG", &contrast, 100, callback_contrast);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，鼠标选中任意区域实现图片裁剪功能。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

Mat img, result;
Point p1, p2, p3, p4;
vector<Point> p;
Rect roi;
bool flag = false;
bool window = false;

void show(int x, int y)
{
    img.copyTo(result);
    if (x < 0)
    {
        x = 0;
    }
    if (y < 0)
    {
        y = 0;
    }
    p3.x = x;
    p3.y = y;
    if (p3.x < p1.x && p3.y < p1.y) // lt
    {
        p2.x = p3.x;
        p2.y = p1.y;
        p4.x = p1.x;
        p4.y = p3.y;
        roi.x = p3.x;
        roi.y = p3.y;
        roi.width = p1.x - p3.x;
        roi.height = p1.y - p3.y;
    }
    else if (p3.x > p1.x && p3.y < p1.y) // rt
    {
        p2.x = p1.x;
        p2.y = p3.y;
        p4.x = p3.x;
        p4.y = p1.y;
        roi.x = p2.x;
        roi.y = p2.y;
        roi.width = p3.x - p1.x;
        roi.height = p1.y - p3.y;
    }
    else if (p3.x > p1.x && p3.y > p1.y) // rb
    {
        p2.x = p3.x;
        p2.y = p1.y;
        p4.x = p1.x;
        p4.y = p3.y;
        roi.x = p1.x;
        roi.y = p1.y;
        roi.width = p3.x - p1.x;
        roi.height = p3.y - p1.y;
    }
    else if (p3.x < p1.x && p3.y > p1.y) // lb
    {
        p2.x = p1.x;
        p2.y = p3.y;
        p4.x = p3.x;
        p4.y = p1.y;
        roi.x = p3.x;
        roi.y = p1.y;
        roi.width = p1.x - p3.x;
        roi.height = p3.y - p1.y;
    }
    else
    {
        return;
    }
    cout << "p1: " << p1 << endl;
    cout << "p2: " << p2 << endl;
    cout << "p3: " << p3 << endl;
    cout << "p4: " << p4 << endl;
    p.clear();
    p.push_back(p1);
    p.push_back(p2);
    p.push_back(p3);
    p.push_back(p4);
    vector<vector<Point>> contours;
    contours.push_back(p);
    drawContours(result, contours, 0, Scalar(255, 255, 255), 1, LINE_AA);
    imshow("IMG", result);
}

void callback_mouse(int event, int x, int y, int flags, void *)
{
    if (event == EVENT_LBUTTONDOWN)
    {
        flag = true;
        imshow("IMG", img);
        p1.x = x;
        p1.y = y;
    }
    else if (flag && event == EVENT_LBUTTONUP)
    {
        show(x, y);
        flag = false;
        if (window)
        {
            destroyWindow("ROI");
        }
        namedWindow("ROI");
        window = true;
        imshow("ROI", img(roi));
    }
    else if (flag && event == EVENT_MOUSEMOVE)
    {
        show(x, y);
    }
}

int main(int argc, char const *argv[])
{
    img = imread("demo.jpg");
    if (img.empty())
    {
        cout << "could not load image." << endl;
        return -1;
    }
    imshow("IMG", img);
    setMouseCallback("IMG", callback_mouse);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，当键盘输入 1 时将原图像转换为 HSV ，当键盘输入 2 时将原图像转换为灰度图，按下键盘 Esc 时程序退出。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img = imread("demo.jpg");
    Mat result = img.clone();
    int k;
    if (img.empty())
    {
        cout << "could not load image." << endl;
        return -1;
    }
    imshow("IMG", img);
    while (true)
    {
        k = waitKey(1);
        if (k == 27)
        {
            break;
        }
        if (k == 49)
        {
            cvtColor(img, result, COLOR_BGR2HSV);
        }
        if (k == 50)
        {
            cvtColor(img, result, COLOR_BGR2GRAY);
        }
        imshow("IMG", result);
    }
    destroyAllWindows();
    return 0;
}
```
