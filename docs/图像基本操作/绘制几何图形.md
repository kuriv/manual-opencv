# 绘制几何图形

执行下面的代码，在图像矩阵上绘制矩形。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img(512, 512, CV_8UC3, Scalar(0, 255, 255));
    Rect rect;
    rect.x = 100;
    rect.y = 100;
    rect.width = 200;
    rect.height = 100;
    rectangle(img, rect, Scalar(255, 255, 0), -1);
    rectangle(img, Point(300, 300), Point(350, 350), Scalar(255, 255, 0), 2);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，在图像矩阵上随机绘制线段。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img = Mat::zeros(Size(512, 512), CV_8UC3);
    img = Scalar(0, 0, 0);
    int rows = img.rows;
    int cols = img.cols;
    int k;
    RNG rng(233);
    int x1, x2, y1, y2;
    int b, g, r;
    while (true)
    {
        img = Scalar(0, 0, 0);
        k = waitKey(100);
        if (k == 27)
        {
            break;
        }
        x1 = rng.uniform(0, rows);
        x2 = rng.uniform(0, rows);
        y1 = rng.uniform(0, cols);
        y2 = rng.uniform(0, cols);
        b = rng.uniform(0, 255);
        g = rng.uniform(0, 255);
        r = rng.uniform(0, 255);
        line(img, Point(x1, y1), Point(x2, y2), Scalar(b, g, r), 1);
        imshow("IMG", img);
    }
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

```cpp
    rr.center = Point(256, 256);
    rr.size = Size(200, 100);
    rr.angle = 0;
    ellipse(img, rr, Scalar(255, 255, 0), -1);
    ellipse(img, Point(100, 50), Size(50, 25), 0, 0, 360, Scalar(255, 255, 0), 2);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

```cpp
#include <opencv2/opencv.hpp>

    vector<Point> pts;
    ellipse2Poly(Point(256, 256), Size(200, 100), 0, 0, 360, 2, pts);
    cout << pts << endl;
    for (int i = 0; i < pts.size() - 1; i++)
    {
        line(img, pts[i], pts[i + 1], Scalar(255, 255, 0), 2, LINE_AA);
    }
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，在图像矩阵上绘制多边形。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img(512, 512, CV_8UC3, Scalar(0, 255, 255));
    Point pp1[2][6];
    pp1[0][0] = Point(100, 100);
    pp1[0][1] = Point(200, 50);
    pp1[0][2] = Point(300, 200);
    pp1[0][3] = Point(250, 300);
    pp1[0][4] = Point(200, 100);
    pp1[0][5] = Point(150, 80);
    pp1[1][0] = Point(360, 340);
    pp1[1][1] = Point(450, 350);
    pp1[1][2] = Point(500, 350);
    pp1[1][3] = Point(480, 420);
    pp1[1][4] = Point(420, 450);
    pp1[1][5] = Point(350, 400);
    Point pp2[5];
    pp2[0] = Point(350, 80);
    pp2[1] = Point(460, 90);
    pp2[2] = Point(500, 170);
    pp2[3] = Point(420, 190);
    pp2[4] = Point(340, 140);
    const Point *pts[] = {pp1[0], pp1[1], pp2};
    int npts[] = {6, 6, 5};
    fillPoly(img, pts, npts, 3, Scalar(255, 255, 0), LINE_AA);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

也可以使用下面的方式绘制并填充多边形。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img(512, 512, CV_8UC3, Scalar(0, 255, 255));
    vector<Point> pp1, pp2, pp3;
    pp1.push_back(Point(100, 100));
    pp1.push_back(Point(200, 50));
    pp1.push_back(Point(300, 200));
    pp1.push_back(Point(250, 300));
    pp1.push_back(Point(200, 100));
    pp1.push_back(Point(150, 80));
    pp2.push_back(Point(360, 340));
    pp2.push_back(Point(450, 350));
    pp2.push_back(Point(500, 350));
    pp2.push_back(Point(480, 420));
    pp2.push_back(Point(420, 450));
    pp2.push_back(Point(350, 400));
    pp3.push_back(Point(350, 80));
    pp3.push_back(Point(460, 90));
    pp3.push_back(Point(500, 170));
    pp3.push_back(Point(420, 190));
    pp3.push_back(Point(340, 140));
    vector<vector<Point>> contours;
    contours.push_back(pp1);
    contours.push_back(pp2);
    contours.push_back(pp3);
    drawContours(img, contours, -1, Scalar(255, 255, 0), -1, LINE_AA);
    imshow("IMG", img);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```
