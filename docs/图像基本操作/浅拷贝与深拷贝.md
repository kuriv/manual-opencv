# 浅拷贝与深拷贝

注意，直接使用赋值操作传递的是图像矩阵的地址引用，即浅拷贝。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img1, img2;
    img1 = Mat::zeros(Size(5, 5), CV_8UC3);
    img1 = Scalar(233, 233, 233);
    img2 = img1;
    img2 = Scalar(0, 0, 0);
    cout << img1 << endl;
    imshow("IMG1", img1);
    cout << img2 << endl;
    imshow("IMG2", img2);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```

执行下面的代码，对图像矩阵进行深拷贝操作。

```cpp
#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main(int argc, char const *argv[])
{
    Mat img1, img2, img3;
    img1 = Mat::zeros(Size(5, 5), CV_8UC3);
    img1 = Scalar(233, 233, 233);
    img1.copyTo(img2);
    copyTo(img1, img3, img1);
    img2 = Scalar(150, 150, 150);
    img3 = Scalar(0, 0, 0);
    cout << img1 << endl;
    imshow("IMG1", img1);
    cout << img2 << endl;
    imshow("IMG2", img2);
    cout << img3 << endl;
    imshow("IMG3", img3);
    waitKey(0);
    destroyAllWindows();
    return 0;
}
```
